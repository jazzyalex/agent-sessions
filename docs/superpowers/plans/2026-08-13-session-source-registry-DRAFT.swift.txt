// MARK: - SessionSourceRegistry

// A DATA-GATHERING PROTOTYPE, not yet wired into the app.
//
// This file collapses the per-source metadata that today is spread across five
// independent switch statements into a single descriptor table:
//   1. AgentSessions/Model/SessionSource.swift            (displayName, iconName, versionIntroduced, featureDescription)
//   2. AgentSessions/Services/TranscriptColorSystem.swift  (agentBrandAccent(source:))
//   3. AgentSessions/Analytics/Utilities/AnalyticsColors.swift (agentXGray monochrome Color(white:))
//   4. AgentSessions/Onboarding/Components/OnboardingComponents.swift (AgentBadge.initials(for:))
//   5. AgentSessions/Views/Preferences/PreferencesConstants.swift + AgentSessions/Services/AgentEnablement.swift
//      (enablement/CLI-available/root-override UserDefaults keys, binary names, default-on behavior)
//
// Every value below is transcribed verbatim from those five files as of this writing.
// Nothing here is invented or "improved" — irregularities found in the source switches
// (missing keys, un-namespaced literals, pass-through vs. wrapped colors, extra keys)
// are preserved and called out in comments rather than papered over.
//
// This file intentionally does NOT add a parseFull/discovery/indexer closure — that is
// a later increment. This is plain per-source value data only.

// MARK: - BrandHue

/// How a source's brand accent color is produced by `TranscriptColorSystem.agentBrandAccent(source:)`.
///
/// This distinction is load-bearing: most sources use a hand-tuned RGB triple that gets
/// wrapped in `adaptiveBrand(_:)` (light value preserved, dark mode gets a brightened/
/// desaturated variant). Two sources (antigravity, opencode) instead pass an AppKit
/// *system dynamic* color straight through, unwrapped — those colors already adapt to
/// appearance on their own, so wrapping them would be redundant.
enum BrandHue {
    /// A system dynamic `NSColor` (e.g. `.systemTeal`), returned unwrapped —
    /// no `adaptiveBrand` treatment, because system colors already adapt.
    case system(NSColor)
    /// A hand-tuned, light-mode-calibrated RGB triple. Reconstituting the actual
    /// `NSColor` requires passing this through `adaptiveBrand(NSColor(calibratedRed:green:blue:alpha:))`
    /// exactly as `TranscriptColorSystem.agentBrandAccent(source:)` does today.
    case calibrated(red: CGFloat, green: CGFloat, blue: CGFloat)
}

// MARK: - SessionSourceDescriptor

/// All per-source metadata that today is scattered across five switch statements,
/// collapsed into one row per `SessionSource`.
struct SessionSourceDescriptor {
    /// The source this descriptor describes.
    let source: SessionSource

    // MARK: Identity (from SessionSource.swift)

    /// Human-facing name, e.g. "Codex CLI".
    let displayName: String
    /// SF Symbol name used for the source's icon.
    let iconName: String
    /// App version in which support for this source was introduced.
    let versionIntroduced: String
    /// One-line description used in onboarding / feature-gate copy.
    let featureDescription: String

    // MARK: Color (from TranscriptColorSystem.swift + AnalyticsColors.swift)

    /// Brand accent hue. See `BrandHue` for the system-passthrough-vs-calibrated distinction.
    let brandHue: BrandHue
    /// Monochrome gray value (`Color(white:)`) used when the analytics view's
    /// "monochrome" mode is on.
    let monochromeWhite: Double

    // MARK: Badge (from OnboardingComponents.swift)

    /// Short label shown on `AgentBadge` (`AgentBadge.initials(for:)`). Two letters,
    /// except droid which is a single "D".
    let badgeAbbreviation: String

    // MARK: UserDefaults keys (from PreferencesConstants.swift + AgentEnablement.swift)

    /// `PreferencesKey.Agents.*Enabled` — every source has one.
    let enablementKey: String
    /// `PreferencesKey.*CLIAvailable` — nil for openclaw, which has no such key at all
    /// (see `AgentEnablement.storedBinaryPresence`, where `.openclaw` returns `nil`
    /// rather than reading a defaults key).
    let cliAvailableKey: String?
    /// `PreferencesKey.Paths.*SessionsRootOverride` — nil only in principle; in practice
    /// every source has *some* root-override key, but two of them (codex, antigravity)
    /// are raw string literals read directly in `AgentEnablement.isAvailable(_:)` rather
    /// than named `PreferencesKey.Paths` constants. Those two are called out per-entry
    /// below since they can't be "referenced by name" the way the other ten can.
    let rootOverrideKey: String?

    // MARK: Binary detection (from AgentEnablement.binaryInstalled(for:))

    /// Executable name(s) checked on PATH. Most sources have exactly one; a few check
    /// multiple aliases.
    let binaryNames: [String]

    // MARK: Default-on behavior (from AgentEnablement.isEnabled(_:))

    /// `true` for sources whose `isEnabled(_:)` case falls into the
    /// "default OFF unless available" group (hermes, openclaw, cursor, pi, kimi, grok);
    /// `false` for sources that hit `isEnabled(_:)`'s `default: return true` branch and
    /// are therefore always default-on regardless of availability (codex, claude,
    /// antigravity, opencode, copilot, droid).
    let defaultsOnOnlyWhenAvailable: Bool
}

    /// Convenience accessor. Every `SessionSource` case has an entry in `all`,
    /// so this force-unwraps rather than returning an optional.
    static func descriptor(for source: SessionSource) -> SessionSourceDescriptor {
        guard let descriptor = all[source] else {
            preconditionFailure("SessionSourceRegistry.all is missing an entry for \(source)")
        }
        return descriptor
    }
